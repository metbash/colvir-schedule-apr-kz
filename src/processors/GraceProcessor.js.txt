/**
 * ==========================================================
 * Colvir Schedule & APR Calculator (KZ)
 * Version: 4.0-dev4
 *
 * GraceProcessor.js
 *
 * Процессор льготного периода.
 *
 * На текущем этапе реализует:
 * - льготу по основному долгу (RULE-002);
 * - льготу по вознаграждению;
 * - полную льготу.
 *
 * Логика подтверждается эталонами Colvir.
 * ==========================================================
 */

import BaseProcessor from "./BaseProcessor.js";
import { GraceType } from "../core/Enums.js";

export default class GraceProcessor extends BaseProcessor {

    /**
     * Проверить тип активной льготы.
     *
     * @param {LoanState} state
     * @returns {string|null}
     */
    process(state) {

        if (state.hasPrincipalGrace()) {

            return GraceType.PRINCIPAL;

        }

        if (state.hasInterestGrace()) {

            return GraceType.INTEREST;

        }

        if (state.hasFullGrace()) {

            return GraceType.FULL;

        }

        return null;

    }

    /**
     * Определить, является ли льгота
     * льготой по основному долгу.
     *
     * @param {LoanState} state
     * @returns {boolean}
     */
    isPrincipalGrace(state) {

        return this.process(state) ===
            GraceType.PRINCIPAL;

    }

    /**
     * Определить, является ли льгота
     * льготой по вознаграждению.
     *
     * @param {LoanState} state
     * @returns {boolean}
     */
    isInterestGrace(state) {

        return this.process(state) ===
            GraceType.INTEREST;

    }

    /**
     * Определить, является ли льгота
     * полной льготой.
     *
     * @param {LoanState} state
     * @returns {boolean}
     */
    isFullGrace(state) {

        return this.process(state) ===
            GraceType.FULL;

    }

}
